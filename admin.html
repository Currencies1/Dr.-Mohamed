async function toggleStatus(id, currentStatus) { const newStatus = currentStatus === 'قيد الانتظار' ? 'مؤكد' : 'قيد الانتظار'; // جلب بيانات الحجز لتفقد الـ fcmToken const doc = await db.collection('appointments').doc(id).get(); const data = doc.data(); await db.collection('appointments').doc(id).update({ status: newStatus }); // إرسال إشعار للخلفية إذا كان المعرف موجوداً والحالة مؤكدة if (newStatus === 'مؤكد' && data.fcmToken) { sendPushNotification(data.fcmToken, data.fullName, data.date); } } function sendPushNotification(token, name, date) { // تتطلب هذه العملية Server Key من لوحة Firebase Cloud Messaging // إرسال الطلب عبر REST API الخاص بـ Firebase fetch('https://fcm.googleapis.com/fcm/send', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': 'key=YOUR_FIREBASE_SERVER_KEY' // ضع هنا Server Key الخاص بمشروعك }, body: JSON.stringify({ to: token, notification: { title: "تم تأكيد الحجز 🎉", body: `أهلاً ${name}، تم تأكيد موعدك بنجاح بتاريخ ${date}.` } }) }); }
